cmake_minimum_required(VERSION 3.16)

# sdkconfig.defaults holds all non-sensitive configuration (committed).
# sdkconfig.secrets holds only WiFi credentials (gitignored); copy
# sdkconfig.secrets.example and fill it in. The build works without it, and the
# firmware then runs with WiFi disabled (USB control unaffected).
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.secrets")
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.secrets")
endif()

# Re-run configuration when the secrets file appears or changes, so the check
# below actually gets a chance to run. Both halves are needed:
#
#  * CMAKE_CONFIGURE_DEPENDS on a path that does NOT yet exist is silently
#    dropped (it never reaches build.ninja), so on its own it misses the
#    "secrets appear" case, which is exactly the case the check below exists for.
#  * A CONFIGURE_DEPENDS glob watches the *directory*, so the file appearing (or
#    being removed) changes the glob result and forces a reconfigure. But a glob
#    only tracks the set of names, so on its own it misses content edits.
file(GLOB _fan_secrets CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.secrets")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_fan_secrets})

# SDKCONFIG_DEFAULTS is only consulted when sdkconfig is *generated*. kconfgen
# loads the defaults first and then layers an existing sdkconfig on top, so once
# sdkconfig exists, sdkconfig.secrets no longer reaches the build. Creating it,
# editing it (a changed password, say) or deleting it all change nothing, and the
# image keeps whatever credentials sdkconfig captured first. The only symptom is
# a log line on a UART that is unconnected on this board. So compare the two by
# content, which is reliable regardless of what touched what when. The values
# are secrets, so the message says which key differs and never what it holds.

# The raw value of a Kconfig line, quotes and escapes included, or "" (the empty
# string Kconfig writes for an unset string) when the key is absent.
function(_fan_kconfig_value text key out)
    string(REGEX MATCH "(^|\n)${key}=[^\n]*" _line "${text}")
    string(REGEX REPLACE "^\n?${key}=" "" _value "${_line}")
    string(STRIP "${_value}" _value)
    if(_value STREQUAL "")
        set(_value "\"\"")
    endif()
    set(${out} "${_value}" PARENT_SCOPE)
endfunction()

if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig")
    file(READ "${CMAKE_CURRENT_LIST_DIR}/sdkconfig" _fan_sdkconfig)
    set(_fan_secrets_text "")
    if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.secrets")
        file(READ "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.secrets" _fan_secrets_text)
    endif()
    foreach(_fan_key CONFIG_FAN_WIFI_SSID CONFIG_FAN_WIFI_PASSWORD)
        _fan_kconfig_value("${_fan_sdkconfig}" ${_fan_key} _fan_have)
        _fan_kconfig_value("${_fan_secrets_text}" ${_fan_key} _fan_want)
        if(NOT _fan_have STREQUAL _fan_want)
            message(FATAL_ERROR
                "${_fan_key} in sdkconfig differs from sdkconfig.secrets (or from "
                "empty, if that file is absent). The existing sdkconfig takes "
                "precedence over SDKCONFIG_DEFAULTS, so this build would silently "
                "keep the credentials sdkconfig captured earlier.\n"
                "Delete it and rebuild:  rm firmware/sdkconfig && idf.py build")
        endif()
    endforeach()
endif()

# The embedded status page is generated by `pnpm build` in frontend/. The
# firmware build depends only on the .gz, never on the sources that produced it,
# so a stale page would otherwise be flashed with no warning, and a zero-byte
# one would build fine and serve a blank page while /api/status still worked.
set(_fan_page "${CMAKE_CURRENT_LIST_DIR}/main/web/index.html.gz")
if(NOT EXISTS "${_fan_page}")
    message(FATAL_ERROR
        "missing ${_fan_page}\nBuild the status page first:  cd frontend && pnpm build")
endif()
file(SIZE "${_fan_page}" _fan_page_size)
if(_fan_page_size LESS 1024)
    message(FATAL_ERROR
        "${_fan_page} is only ${_fan_page_size} bytes -- the page did not build "
        "correctly.\nRebuild it:  cd frontend && pnpm build")
endif()
# Every input that changes the built bundle. .env.development is deliberately
# absent: it is not baked into a production build, so editing it must not fail
# the firmware build.
set(_fan_frontend "${CMAKE_CURRENT_LIST_DIR}/../frontend")
# Recursive only under src/, which contains nothing but page sources.
file(GLOB_RECURSE _fan_page_sources CONFIGURE_DEPENDS "${_fan_frontend}/src/*")
# Non-recursive, and that matters: GLOB_RECURSE with an exact filename recurses
# from the directory part, so "frontend/package.json" also matches every
# node_modules/**/package.json, dozens of them. That would flood the dependency
# list and, far worse, make the staleness check below fail on every build,
# because pnpm's install timestamps are newer than the page.
file(GLOB _fan_build_inputs CONFIGURE_DEPENDS
    "${_fan_frontend}/index.html"
    "${_fan_frontend}/vite.config.js"
    "${_fan_frontend}/package.json"
    "${_fan_frontend}/pnpm-lock.yaml"
    "${_fan_frontend}/.env.production"
    "${_fan_frontend}/scripts/embed.mjs")
list(APPEND _fan_page_sources ${_fan_build_inputs})

# The glob above re-runs CMake when a source is added or removed, but NOT when
# one is edited in place: CONFIGURE_DEPENDS on a glob tracks the set of matching
# names, not their contents. Without this line the common case (edit the page,
# rebuild the firmware, forget `pnpm build`) would skip the check entirely and
# flash the stale page, which is the one thing the check is for.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${_fan_page_sources} "${_fan_page}")

foreach(_src IN LISTS _fan_page_sources)
    # Phrased as "the page is NOT newer than the source" rather than "the source
    # is newer than the page", because IS_NEWER_THAN reports true for an exact
    # timestamp tie. Testing the source directly therefore failed a build whose
    # page was merely written in the same clock tick as a source -- reachable on
    # a filesystem with 1-second granularity, and after any copy that preserves
    # mtimes. This form treats a tie as up to date.
    if(NOT "${_fan_page}" IS_NEWER_THAN "${_src}")
        message(FATAL_ERROR
            "${_src} is newer than the embedded status page, which would flash a "
            "stale UI.\nRebuild it:  cd frontend && pnpm build")
    endif()
endforeach()

# The product version is shared with the daemon and the status page, so it is read
# from the repository's VERSION file rather than restated here. PROJECT_VER lands
# in the app descriptor, which means a flashed image can be identified with
# esptool instead of guessed at.
#
# PROTOCOL_VERSION in main/constants.h is deliberately separate: that one is the
# wire contract the daemon gates on, and it must only change when the protocol
# does.
set(_fan_version_file "${CMAKE_CURRENT_LIST_DIR}/../VERSION")
if(NOT EXISTS "${_fan_version_file}")
    message(FATAL_ERROR "missing ${_fan_version_file}, which carries the product version")
endif()
file(READ "${_fan_version_file}" _fan_version)
string(STRIP "${_fan_version}" _fan_version)
set(PROJECT_VER "${_fan_version}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_fan_version_file}")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(fan-controller)
